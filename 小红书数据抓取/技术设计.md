# 技术设计 - 最佳实践重构

## 一、架构设计

### 1.1 模块划分

```
xhs-to-notion.user.js
├── ConfigManager        # 配置管理
├── NotionAPI           # Notion API 封装
├── NoteExtractor       # 笔记数据提取
├── AccountExtractor    # 账号数据提取
├── UIManager           # UI 管理
└── SyncController      # 同步控制器
```

### 1.2 设计原则

- **单一职责**：每个模块只负责一个功能
- **依赖注入**：模块间通过接口通信
- **错误隔离**：每个模块独立处理错误
- **可测试性**：纯函数优先，副作用隔离

### 2.6 SyncController（同步控制器）

**职责**：协调各模块，控制同步流程

```javascript
class SyncController {
  constructor(config, notionAPI, extractor, uiManager) {
    this.config = config;
    this.notionAPI = notionAPI;
    this.extractor = extractor;
    this.uiManager = uiManager;
  }

  async syncNote() {
    // 1. 提取数据
    const data = this.extractor.extract();

    // 2. 查询是否存在
    const existing = await this.notionAPI.findByUniqueField('URL', data.url);

    // 3. 显示确认对话框
    const confirmed = await this.uiManager.showConfirmDialog(
      existing ? 'update' : 'create',
      data,
      existing
    );

    if (!confirmed) return;

    // 4. 执行同步
    if (existing) {
      await this.notionAPI.updatePage(existing.id, data);
    } else {
      await this.notionAPI.createPage(data);
    }

    // 5. 显示结果
    this.uiManager.showNotification('同步成功', 'success');
  }

  async syncAccount() { /* 类似流程 */ }

  async initialize() {
    // 初始化数据库结构
    await this.notionAPI.ensureSchema(REQUIRED_FIELDS);
  }
}
```

**关键点**：
- 统一的同步流程
- 完整的错误处理
- 用户友好的反馈

## 四、关键技术实现

### 4.1 去重实现

```javascript
async findByUniqueField(fieldName, value) {
  const result = await this.queryDatabase({
    filter: {
      property: fieldName,
      [this.getFilterType(fieldName)]: { equals: value }
    }
  });

  return result.results[0] || null;
}
```

### 4.2 字段自动创建

```javascript
async ensureSchema(schema) {
  const db = await this.getDatabase();
  const existing = db.properties;
  const missing = {};

  for (const [name, config] of Object.entries(schema)) {
    if (!existing[name]) {
      missing[name] = { [config.type]: {} };
    }
  }

  if (Object.keys(missing).length > 0) {
    await this.updateDatabaseSchema(missing);
  }
}
```

### 4.3 图片处理

```javascript
extractImages() {
  const imgs = document.querySelectorAll('.note-content img');
  return Array.from(imgs)
    .map(img => img.src)
    .filter(src => src && !src.includes('placeholder'))
    .map(src => ({ type: 'external', external: { url: src } }));
}
```

### 4.4 账号笔记列表采集

```javascript
async extractNoteList() {
  const notes = [];
  let lastHeight = 0;
  let retries = 0;

  while (retries < 5) {
    // 滚动到底部
    window.scrollTo(0, document.body.scrollHeight);
    await this.sleep(1000);

    // 提取笔记
    const items = document.querySelectorAll('.note-item');
    for (const item of items) {
      const title = item.querySelector('.title')?.textContent;
      const likes = item.querySelector('.likes')?.textContent;
      if (title && !notes.find(n => n.title === title)) {
        notes.push({ title, likes });
      }
    }

    // 检查是否到底
    if (document.body.scrollHeight === lastHeight) {
      retries++;
    } else {
      retries = 0;
    }
    lastHeight = document.body.scrollHeight;
  }

  return notes;
}

formatNoteList(notes) {
  return notes
    .map(n => `📝 ${n.title} | ❤️ ${n.likes}`)
    .join('\n');
}
```

## 五、初始化流程

```javascript
(function() {
  'use strict';

  // 1. 创建实例
  const config = new ConfigManager();
  const uiManager = new UIManager();

  // 2. 注入样式
  uiManager.injectStyles();

  // 3. 检查配置
  if (!config.isConfigured('note')) {
    config.showConfigUI();
    return;
  }

  // 4. 创建 API 实例
  const noteAPI = new NotionAPI(
    config.config.noteApiKey,
    config.config.noteDatabaseId
  );

  const accountAPI = new NotionAPI(
    config.config.accountApiKey,
    config.config.accountDatabaseId
  );

  // 5. 初始化数据库
  Promise.all([
    noteAPI.ensureSchema(NOTE_SCHEMA),
    accountAPI.ensureSchema(ACCOUNT_SCHEMA)
  ]).then(() => {
    console.log('[XHS] 初始化完成');

    // 6. 创建控制器
    const noteController = new SyncController(
      config,
      noteAPI,
      new NoteExtractor(),
      uiManager
    );

    const accountController = new SyncController(
      config,
      accountAPI,
      new AccountExtractor(),
      uiManager
    );

    // 7. 注入按钮
    uiManager.startObserver(() => {
      if (isNotePage()) {
        uiManager.createNoteButton(() => noteController.syncNote());
      } else if (isAccountPage()) {
        uiManager.createAccountButton(() => accountController.syncAccount());
      }
    });
  });

  // 8. 注册菜单命令
  GM_registerMenuCommand('配置笔记数据库', () => config.showConfigUI('note'));
  GM_registerMenuCommand('配置账号数据库', () => config.showConfigUI('account'));
})();
```

## 六、错误处理策略

### 6.1 分层错误处理

```javascript
// 提取器层：返回默认值
extractTitle() {
  try {
    return document.querySelector('.title')?.textContent || '未知标题';
  } catch (e) {
    console.warn('提取标题失败', e);
    return '未知标题';
  }

// API 层：抛出错误
async createPage(data) {
  try {
    const response = await this.request('POST', '/pages', data);
    return response;
  } catch (e) {
    throw new Error(`创建页面失败: ${e.message}`);
  }
}

// 控制器层：用户提示
async syncNote() {
```

## 七、性能优化

### 7.1 缓存策略

```javascript
class NotionAPI {
  constructor(apiKey, databaseId) {
    this.apiKey = apiKey;
    this.databaseId = databaseId;
    this.schemaCache = null; // 缓存数据库结构
  }

  async getDatabase() {
    if (this.schemaCache) return this.schemaCache;
    this.schemaCache = await this.fetchDatabase();
    return this.schemaCache;
  }
}
```

### 7.2 防抖处理

```javascript
class UIManager {
  createNoteButton(onClick) {
    const btn = document.createElement('button');
    btn.onclick = this.debounce(onClick, 1000);
    return btn;
  }

  debounce(fn, delay) {
    let timer = null;
    return function(...args) {
      if (timer) return;
      timer = setTimeout(() => timer = null, delay);
      return fn.apply(this, args);
    };
  }
}
```
